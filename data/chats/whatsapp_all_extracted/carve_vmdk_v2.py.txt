import os
import struct
import sqlite3

def scan_file(vmdk_path, out_dir):
    magic = b'SQLite format 3\x00'
    found = 0
    file_size = os.path.getsize(vmdk_path)
    print(f"\n--- Escaneando {vmdk_path} ({file_size / (1024*1024):.1f} MB) ---")

    with open(vmdk_path, 'rb') as f:
        chunk_size = 64 * 1024 * 1024
        pos = 0
        while True:
            chunk = f.read(chunk_size)
            if not chunk:
                break
            idx = 0
            while True:
                idx = chunk.find(magic, idx)
                if idx == -1:
                    break
                abs_pos = pos + idx
                
                # Leer cabecera SQLite
                f.seek(abs_pos)
                hdr = f.read(100)
                if len(hdr) == 100 and hdr[:16] == magic:
                    page_size = struct.unpack('>H', hdr[16:18])[0]
                    if page_size == 1:
                        page_size = 65536
                    page_count = struct.unpack('>I', hdr[28:32])[0]
                    
                    # Si page_count es 0 en WAL mode, estimamos tamaño
                    if page_count == 0:
                        page_count = 5000
                    
                    if 512 <= page_size <= 65536 and 1 <= page_count <= 2000000:
                        total_size = min(page_size * page_count, 100 * 1024 * 1024)
                        if total_size > 10 * 1024:  # mas de 10 KB
                            f.seek(abs_pos)
                            db_bytes = f.read(total_size)
                            temp_file = os.path.join(out_dir, f'temp_{hex(abs_pos)}.db')
                            with open(temp_file, 'wb') as tf:
                                tf.write(db_bytes)
                            
                            try:
                                c = sqlite3.connect(temp_file)
                                cur = c.cursor()
                                cur.execute("SELECT name FROM sqlite_master WHERE type='table'")
                                tables = [r[0] for r in cur.fetchall()]
                                c.close()
                                
                                is_wa = any(t in tables for t in ['message', 'chat', 'jid', 'wa_contacts', 'messages', 'message_view', 'settings', 'user_device'])
                                if is_wa:
                                    print(f"🎉 [ENCONTRADA] DB WhatsApp en offset {hex(abs_pos)}! Tamano: {len(db_bytes)/(1024*1024):.2f} MB")
                                    print(f"   Tablas: {tables}")
                                    
                                    dest = os.path.join(out_dir, f'whatsapp_db_{hex(abs_pos)}.db')
                                    if os.path.exists(dest):
                                        os.remove(dest)
                                    os.rename(temp_file, dest)
                                    found += 1
                                    
                                    if 'message' in tables or 'messages' in tables:
                                        target_msgstore = r'c:\Users\mayra\projects\Medicina\whatsapp_business_backup\msgstore.db'
                                        with open(target_msgstore, 'wb') as mf:
                                            mf.write(db_bytes)
                                        print(f"   ⭐⭐⭐ GUARDADA COMO MSGSTORE.DB PRINCIPAL! ⭐⭐⭐")
                                else:
                                    if os.path.exists(temp_file):
                                        os.remove(temp_file)
                            except Exception:
                                if os.path.exists(temp_file):
                                    os.remove(temp_file)

                f.seek(pos + idx + 16)
                idx += 16
            pos += len(chunk)

    print(f"Total bases encontradas en este archivo: {found}")
    return found

def main():
    out_dir = r'c:\Users\mayra\projects\Medicina\whatsapp_business_backup\extracted_dbs'
    os.makedirs(out_dir, exist_ok=True)
    
    vms = [
        r'C:\LDPlayer\LDPlayer9\vms\leidian0\data.vmdk',
        r'C:\LDPlayer\LDPlayer9\vms\leidian0\sdcard.vmdk',
        r'C:\LDPlayer\LDPlayer9\data.vmdk',
    ]
    for v in vms:
        if os.path.exists(v):
            scan_file(v, out_dir)

if __name__ == "__main__":
    main()
